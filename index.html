const fs = require('fs'); const TelegramBot = require('node-telegram-bot-api'); const express = require('express'); const cors = require('cors'); const token = process.env.BOT_TOKEN; const OWNER_ID = '1159481047'; // YOU - Mnelisi const bot = new TelegramBot(token, { polling: true }); const app = express(); app.use(cors()); app.use(express.json()); const APP_URL = 'https://mnelisimaseko454-boop.github.io/ShayaWatchMnelisibot/'; const USER_EARN = 0.10; // User gets 0.10, YOU keep R1.40 profit const MIN_WITHDRAW = 10.00; const DB_FILE = './balances.json'; const WD_FILE = './withdrawals.json'; let balances = {}; let withdrawRequests = []; if (fs.existsSync(DB_FILE)) { try { balances = JSON.parse(fs.readFileSync(DB_FILE, 'utf8')); } catch(e){} } if (fs.existsSync(WD_FILE)) { try { withdrawRequests = JSON.parse(fs.readFileSync(WD_FILE, 'utf8')); } catch(e){} } function saveDB() { fs.writeFileSync(DB_FILE, JSON.stringify(balances, null, 2)); } function saveWD() { fs.writeFileSync(WD_FILE, JSON.stringify(withdrawRequests, null, 2)); } bot.onText(/\/start/, (msg) => { const userId = String(msg.from.id); if (!balances[userId]) { balances[userId] = { balance: 0, totalEarned: 0, username: msg.from.username || msg.from.first_name || 'user' }; saveDB(); } bot.sendMessage(msg.chat.id, `🎵 SHAYA WATCH\n\nYour Balance: R${balances[userId].balance.toFixed(2)}\nEarn R${USER_EARN.toFixed(2)} per ad\nMin withdraw: R${MIN_WITHDRAW}\n\nClick to open:`, { reply_markup: { inline_keyboard: [[{ text: "▶️ OPEN SHAYA WATCH", url: APP_URL }]] } }); }); // API: Get balance - NEVER resets when clear phone app.get('/api/balance', (req, res) => { const userId = String(req.query.userId || ''); if (!userId) return res.json({ balance: 0 }); const data = balances[userId] || { balance: 0, totalEarned: 0 }; res.json(data); }); // API: Earn app.post('/api/earn', (req, res) => { const userId = String(req.body.userId || ''); const username = req.body.username || 'user'; if (!userId) return res.json({ error: 'no user' }); if (!balances[userId]) balances[userId] = { balance: 0, totalEarned: 0, username }; balances[userId].balance += USER_EARN; balances[userId].totalEarned += USER_EARN; balances[userId].username = username; saveDB(); res.json(balances[userId]); }); // API: Withdraw app.post('/api/withdraw', (req, res) => { const userId = String(req.body.userId || ''); const { method, account } = req.body; const user = balances[userId]; if (!user || user.balance < MIN_WITHDRAW) { return res.json({ error: `Need R${MIN_WITHDRAW} minimum. You have R${user ? user.balance.toFixed(2) : '0'}` }); } const amount = user.balance; withdrawRequests.push({ userId, username: user.username, method, account, amount, date: new Date().toISOString() }); saveWD(); user.balance = 0; saveDB(); // Alert YOU bot.sendMessage(OWNER_ID, `💰 NEW WITHDRAWAL REQUEST!\n\nUser: @${user.username} (${userId})\nAmount: R${amount.toFixed(2)}\nMethod: ${method}\nAccount: ${account}\n\nPay them! Then type /paid ${userId}`); res.json({ success: true, message: `Withdrawal R${amount.toFixed(2)} requested! Paid in 24h to ${method}: ${account}` }); }); bot.onText(/\/paid (.+)/, (msg, match) => { if (String(msg.from.id) !== OWNER_ID) return; const userId = String(match[1]); bot.sendMessage(userId, `✅ Your withdrawal has been PAID! Check your account. Thank you for using Shaya Watch 🎵`); bot.sendMessage(OWNER_ID, `Marked ${userId} as paid.`); }); app.get('/', (req, res) => res.send(`Shaya Watch API running. Balances: ${Object.keys(balances).length}`)); const PORT = process.env.PORT || 10000; app.listen(PORT, () => console.log(`API running on ${PORT}`));
